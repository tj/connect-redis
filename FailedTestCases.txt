
test("session expiration with zero TTL", async () => {
  let client = createClient({url: `redis://localhost:${redisSrv.port}`})
  await client.connect()
  let store = new RedisStore({client})

  const P = (f: any) => promisify(f).bind(store)
  let expires = new Date(Date.now()).toISOString()
  await P(store.set)("789", {cookie: {expires}})

  let res = await P(store.length)()
  expect(res).toBe(0) // no key remains

  await client.disconnect()
})

test("zero TTL session destruction", async () => {
  let client = createClient({url: `redis://localhost:${redisSrv.port}`})
  await client.connect()
  let store = new RedisStore({client})

  const P = (f: any) => promisify(f).bind(store)
  const sid = "zeroTTL"
  const sess = {cookie: {expires: new Date(Date.now()).toISOString()}}

  await P(store.set)(sid, sess)

  let res = await P(store.get)(sid)
  expect(res).toBeNull()

  await client.disconnect()
})

test("touch with disableTouch set to true", async () => {
  let client = createClient({url: `redis://localhost:${redisSrv.port}`})
  await client.connect()
  let store = new RedisStore({client, disableTouch: true})

  const P = (f: any) => promisify(f).bind(store)
  let sess = {foo: "bar"}
  await P(store.set)("touchDisabled", sess)

  let ttlBefore = await client.ttl("sess:touchDisabled")
  await P(store.touch)("touchDisabled", sess)
  let ttlAfter = await client.ttl("sess:touchDisabled")

  expect(ttlBefore).toBe(ttlAfter)

  await client.disconnect()
})
